from pymodbus.client import ModbusTcpClient
import time

# UR 로봇 연결
client = ModbusTcpClient('192.168.1.7', port=502)

if client.connect():
    print("✓ Modbus TCP 연결 성공")
    
    # 로봇 → 펜던트 데이터 읽기 (ADDR 128~160)
    result = client.read_input_registers(address=128, count=33, unit=1)
    
    if not result.isError():
        # 하트비트
        heartbeat = result.registers[0]
        
        # 프로그램 동작 여부
        program_running = result.registers[1]
        
        # 용접 상태
        welding_active = result.registers[2]
        
        # 전류/전압 피드백
        current_fb = result.registers[3]  # A
        voltage_fb = result.registers[4] * 0.1  # V
        
        # PATH 정보
        current_path_cell = result.registers[7]
        current_path = result.registers[8]
        
        # 로봇 상태
        robot_ready = result.registers[13]
        robot_error = result.registers[14]
        
        print(f"\n=== 로봇 상태 ===")
        print(f"하트비트: {heartbeat}")
        print(f"프로그램 동작: {'ON' if program_running else 'OFF'}")
        print(f"용접 상태: {'용접중' if welding_active else '대기'}")
        print(f"전류: {current_fb}A")
        print(f"전압: {voltage_fb}V")
        print(f"현재 PATH: {current_path}")
        print(f"로봇 준비: {'Yes' if robot_ready else 'No'}")
        print(f"로봇 에러: {'발생' if robot_error else '정상'}")
    else:
        print(f"✗ 읽기 에러: {result}")
    
    client.close()
else:
    print("✗ 연결 실패")

2. 용접 모니터링 시스템

2.1 실시간 모니터링 클래스

from pymodbus.client import ModbusTcpClient
import time
import math
from datetime import datetime

class WeldingMonitor:
    """조선소 용접 시스템 실시간 모니터링 클래스"""
    
    def __init__(self, robot_ip='192.168.1.7', port=502):
        self.robot_ip = robot_ip
        self.port = port
        self.client = None
        self.running = False
        self.last_heartbeat = 0
        
    def connect(self):
        """Modbus TCP 연결"""
        self.client = ModbusTcpClient(self.robot_ip, port=self.port)
        return self.client.connect()
    
    def disconnect(self):
        """연결 해제"""
        if self.client:
            self.client.close()
    
    def read_robot_to_pendant(self):
        """로봇 → 펜던트 데이터 읽기"""
        result = self.client.read_input_registers(address=128, count=33, unit=1)
        
        if result.isError():
            return None
        
        return {
            'heartbeat': result.registers[0],
            'program_running': bool(result.registers[1]),
            'welding_active': bool(result.registers[2]),
            'current_feedback': result.registers[3],
            'voltage_feedback': result.registers[4] * 0.1,
            'current_setpoint': result.registers[5],
            'voltage_setpoint': result.registers[6] * 0.1,
            'current_path_cell': result.registers[7],
            'current_path': result.registers[8],
            'welding_finished': bool(result.registers[9]),
            'total_current_path': result.registers[10],
            'total_paths': result.registers[11],
            'robot_ready': bool(result.registers[13]),
            'robot_error': bool(result.registers[14]),
            'current_time': result.registers[15],
            'total_time': result.registers[16],
        }
    
    def read_robot_status(self):
        """로봇 시스템 상태 읽기"""
        # 로봇 모드
        result = self.client.read_input_registers(address=258, count=1, unit=1)
        robot_mode = result.registers[0] if not result.isError() else -1
        
        # 전원 및 안전 상태
        result = self.client.read_input_registers(address=260, count=6, unit=1)
        if result.isError():
            return None
        
        return {
            'robot_mode': robot_mode,
            'power_on': bool(result.registers[0]),
            'security_stopped': bool(result.registers[1]),
            'emergency_stopped': bool(result.registers[2]),
            'safety_stopped': bool(result.registers[5]),
        }
    
    def read_joint_data(self):
        """조인트 데이터 읽기"""
        # 조인트 각도
        result = self.client.read_input_registers(address=270, count=6, unit=1)
        if result.isError():
            return None
        
        angles_mrad = result.registers
        angles_deg = [angle / 1000.0 * (180.0 / math.pi) for angle in angles_mrad]
        
        # 조인트 전류
        result = self.client.read_input_registers(address=290, count=6, unit=1)
        currents_ma = result.registers if not result.isError() else [0]*6
        currents_a = [current / 1000.0 for current in currents_ma]
        
        # 조인트 온도
        result = self.client.read_input_registers(address=300, count=6, unit=1)
        temps = result.registers if not result.isError() else [0]*6
        
        return {
            'angles': angles_deg,
            'currents': currents_a,
            'temperatures': temps
        }
    
    def check_heartbeat(self, current_heartbeat):
        """하트비트 확인"""
        if current_heartbeat == self.last_heartbeat:
            return False  # 하트비트 정지
        self.last_heartbeat = current_heartbeat
        return True
    
    def display_status(self, data, status, joints):
        """상태 출력"""
        print("\n" + "="*80)
        print(f"[{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}] 조선소 용접 시스템 모니터링")
        print("="*80)
        
        # 로봇 모드
        modes = {0: "Disconnected", 1: "Confirm Safety", 2: "Booting",
                 3: "Power Off", 5: "Idle", 6: "Backdrive", 7: "Running"}
        mode_name = modes.get(status['robot_mode'], 'Unknown')
        
        print(f"\n[로봇 상태]")
        print(f"  모드: {mode_name}")
        print(f"  전원: {'ON' if status['power_on'] else 'OFF'}")
        print(f"  하트비트: {data['heartbeat']} {'✓' if self.check_heartbeat(data['heartbeat']) else '✗ 정지'}")
        
        # 안전 상태
        if status['emergency_stopped']:
            print(f"  ⚠️  비상 정지 활성화!")
        if status['security_stopped']:
            print(f"  ⚠️  보안 정지 활성화!")
        if status['safety_stopped']:
            print(f"  ⚠️  안전 정지 활성화!")
        
        # 용접 상태
        print(f"\n[용접 상태]")
        print(f"  프로그램: {'실행중' if data['program_running'] else '정지'}")
        print(f"  용접: {'진행중' if data['welding_active'] else '대기'}")
        print(f"  전류: {data['current_feedback']}A (설정: {data['current_setpoint']}A)")
        print(f"  전압: {data['voltage_feedback']:.1f}V (설정: {data['voltage_setpoint']:.1f}V)")
        
        # PATH 정보
        print(f"\n[PATH 정보]")
        print(f"  현재 셀: {data['current_path_cell']}")
        print(f"  현재 PATH: {data['current_path']}")
        print(f"  전체 진행: {data['total_current_path']}/{data['total_paths']}")
        print(f"  진행 시간: {data['current_time']}s / {data['total_time']}s")
        
        # 조인트 상태
        if joints:
            print(f"\n[조인트 상태]")
            for i in range(6):
                temp_status = "⚠️" if joints['temperatures'][i] > 60 else "✓"
                print(f"  Joint {i+1}: {joints['angles'][i]:7.2f}° | "
                      f"{joints['currents'][i]:5.2f}A | "
                      f"{temp_status} {joints['temperatures'][i]}°C")
        
        # 에러 확인
        if data['robot_error']:
            print(f"\n⚠️  로봇 에러 발생!")
    
    def monitor(self, interval=1.0):
        """실시간 모니터링 시작"""
        self.running = True
        print("모니터링 시작... (Ctrl+C로 종료)")
        
        try:
            while self.running:
                # 데이터 읽기
                data = self.read_robot_to_pendant()
                status = self.read_robot_status()
                joints = self.read_joint_data()
                
                if data and status:
                    self.display_status(data, status, joints)
                else:
                    print("✗ 데이터 읽기 실패")
                
                time.sleep(interval)
        
        except KeyboardInterrupt:
            print("\n\n모니터링 종료")
            self.running = False

# 사용 예제
if __name__ == "__main__":
    monitor = WeldingMonitor('192.168.1.7')
    
    if monitor.connect():
        print("✓ UR 로봇 연결 성공")
        monitor.monitor(interval=2.0)  # 2초마다 업데이트
        monitor.disconnect()
    else:
        print("✗ 연결 실패")

3.1 용접 조건 설정

from pymodbus.client import ModbusTcpClient

def set_welding_conditions(client, current, voltage, dimensions):
    """
    용접 조건 및 치수 정보 설정
    
    Args:
        client: ModbusTcpClient 인스턴스
        current: 전류 (A)
        voltage: 전압 (V)
        dimensions: 치수 정보 딕셔너리 {'vertical_left': mm, 'vertical_right': mm, 'horizontal': mm}
    """
    # 1. 작업 모드 설정 (자동 모드)
    result = client.write_register(address=162, value=2, unit=1)
    if result.isError():
        print("✗ 작업 모드 설정 실패")
        return False
    
    # 2. 용접 조건 설정 (전압, 전류)
    voltage_scaled = int(voltage * 10)  # 0.1V 단위로 스케일링
    
    welding_data = [
        voltage_scaled,  # ADDR 231: 전압
        current,         # ADDR 232: 전류
    ]
    
    result = client.write_registers(address=231, values=welding_data, unit=1)
    if result.isError():
        print("✗ 용접 조건 설정 실패")
        return False
    
    # 3. 치수 정보 설정
    dimension_data = [
        dimensions['vertical_left'],   # ADDR 173: 수직좌
        dimensions['vertical_right'],  # ADDR 174: 수직우
        dimensions['horizontal'],      # ADDR 175: 수평 바닥
    ]
    
    result = client.write_registers(address=173, values=dimension_data, unit=1)
    if result.isError():
        print("✗ 치수 정보 설정 실패")
        return False
    
    print("✓ 용접 조건 및 치수 정보 설정 완료")
    print(f"  전류: {current}A")
    print(f"  전압: {voltage}V")
    print(f"  수직좌: {dimensions['vertical_left']}mm")
    print(f"  수직우: {dimensions['vertical_right']}mm")
    print(f"  수평: {dimensions['horizontal']}mm")
    
    return True

# 사용 예제
client = ModbusTcpClient('192.168.1.7', port=502)

if client.connect():
    # 용접 조건 설정
    set_welding_conditions(
        client,
        current=180,
        voltage=25.5,
        dimensions={
            'vertical_left': 150,
            'vertical_right': 200,
            'horizontal': 100
        }
    )
    
    client.close()

3.2 용접 파라미터 설정

def set_welding_parameters(client, params):
    """
    용접 파라미터 설정
    
    Args:
        params: 파라미터 딕셔너리
    """
    data = []
    
    # 시작 파라미터 (ADDR 241~245)
    data.extend([
        params.get('start_weld_time', 0),      # 시작 용접시간 (ms)
        params.get('start_gas_time', 0),       # 시작 가스 (ms)
        params.get('start_backtrack', 0),      # 시작 후진 거리 (mm)
        params.get('start_current', 0),        # 시작 전류 (A)
        int(params.get('start_voltage', 0) * 10),  # 시작 전압 (V)
    ])
    
    # 종료 파라미터 1단계 (ADDR 246~250)
    data.extend([
        params.get('end_weld_time', 0),        # 끝 용접시간 (ms)
        params.get('end_gas_time', 0),         # 끝 가스 (ms)
        params.get('end_backtrack', 0),        # 끝 후진 거리 (mm)
        params.get('end_current', 0),          # 끝 전류 (A)
        int(params.get('end_voltage', 0) * 10),    # 끝 전압 (V)
    ])
    
    # 종료 파라미터 2단계 (ADDR 251~255)
    data.extend([
        params.get('end_weld_time2', 0),       # 끝 용접시간2 (ms)
        params.get('end_gas_time2', 0),        # 끝 가스2 (ms)
        params.get('end_backtrack2', 0),       # 끝 후진 거리2 (mm)
        params.get('end_current2', 0),         # 끝 전류2 (A)
        int(params.get('end_voltage2', 0) * 10),   # 끝 전압2 (V)
    ])
    
    # 데이터 쓰기 (ADDR 241~255)
    result = client.write_registers(address=241, values=data, unit=1)
    
    if not result.isError():
        print("✓ 용접 파라미터 설정 완료")
        return True
    else:
        print(f"✗ 파라미터 설정 실패: {result}")
        return False

# 사용 예제
client = ModbusTcpClient('192.168.1.7', port=502)

using System;
using System.Net.Sockets;
using Modbus.Device;

namespace ShipyardWelding
{
    class ModbusExample
    {
        static void Main(string[] args)
        {
            string ipAddress = "192.168.1.7";
            int port = 502;
            byte unitId = 1;

            using (TcpClient tcpClient = new TcpClient(ipAddress, port))
            {
                ModbusIpMaster master = ModbusIpMaster.CreateIp(tcpClient);

                try
                {
                    Console.WriteLine("✓ Modbus TCP 연결 성공\n");

                    // 1. 로봇 → 펜던트 데이터 읽기 (ADDR 128~160)
                    ushort startAddress = 128;
                    ushort numRegisters = 33;
                    ushort[] registers = master.ReadInputRegisters(
                        unitId, startAddress, numRegisters);

                    Console.WriteLine("=== 로봇 상태 ===");
                    Console.WriteLine($"하트비트: {registers[0]}");
                    Console.WriteLine($"프로그램 동작: {(registers[1] == 1 ? "ON" : "OFF")}");
                    Console.WriteLine($"용접 상태: {(registers[2] == 1 ? "용접중" : "대기")}");
                    Console.WriteLine($"전류: {registers[3]}A");
                    Console.WriteLine($"전압: {registers[4] * 0.1}V");
                    Console.WriteLine($"로봇 에러: {(registers[14] == 1 ? "발생" : "정상")}");

                    // 2. 펜던트 → 로봇 데이터 쓰기 (작업 모드 설정)
                    ushort modeAddress = 162;
                    ushort autoMode = 2;
                    master.WriteSingleRegister(unitId, modeAddress, autoMode);
                    Console.WriteLine("\n✓ 작업 모드: 자동 모드로 설정");

                    // 3. 용접 조건 설정 (전류, 전압)
                    ushort voltageAddress = 231;
                    ushort[] weldingData = new ushort[] 
                    {
                        255,  // 전압 25.5V
                        180   // 전류 180A
                    };
                    master.WriteMultipleRegisters(unitId, voltageAddress, weldingData);
                    Console.WriteLine("✓ 용접 조건 설정 완료");
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"✗ 에러 발생: {ex.Message}");
                }
            }

            Console.WriteLine("\n프로그램 종료");
            Console.ReadKey();
        }
    }
}

4.3 실시간 모니터링 클래스

using System;
using System.Net.Sockets;
using System.Threading;
using Modbus.Device;

namespace ShipyardWelding
{
    public class WeldingMonitor
    {
        private string ipAddress;
        private int port;
        private byte unitId;
        private bool running;

        public WeldingMonitor(string ip, int portNum = 502, byte unit = 1)
        {
            ipAddress = ip;
            port = portNum;
            unitId = unit;
            running = false;
        }

        public void StartMonitoring(int intervalMs = 2000)
        {
            running = true;
            Console.WriteLine("모니터링 시작... (Ctrl+C로 종료)\n");

            using (TcpClient tcpClient = new TcpClient(ipAddress, port))
            {
                ModbusIpMaster master = ModbusIpMaster.CreateIp(tcpClient);

                while (running)
                {
                    try
                    {
                        // 로봇 → 펜던트 데이터 읽기
                        ushort[] data = master.ReadInputRegisters(unitId, 128, 33);

                        // 로봇 상태 읽기
                        ushort[] status = master.ReadInputRegisters(unitId, 258, 8);

                        DisplayStatus(data, status);

                        Thread.Sleep(intervalMs);
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"✗ 에러: {ex.Message}");
                        Thread.Sleep(intervalMs);
                    }
                }
            }
        }

        private void DisplayStatus(ushort[] data, ushort[] status)
        {
            Console.Clear();
            Console.WriteLine("="+ new string('=', 79));
            Console.WriteLine($"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] 조선소 용접 시스템 모니터링");
            Console.WriteLine("=" + new string('=', 79));

            // 로봇 모드
            string[] modes = { "Disconnected", "Confirm Safety", "Booting", 
                             "Power Off", "", "Idle", "Backdrive", "Running" };
            string modeName = status[0] < modes.Length ? modes[status[0]] : "Unknown";

            Console.WriteLine("\n[로봇 상태]");
            Console.WriteLine($"  모드: {modeName}");
            Console.WriteLine($"  하트비트: {data[0]}");
            Console.WriteLine($"  전원: {(status[2] == 1 ? "ON" : "OFF")}");

            // 용접 상태
            Console.WriteLine("\n[용접 상태]");
            Console.WriteLine($"  프로그램: {(data[1] == 1 ? "실행중" : "정지")}");
            Console.WriteLine($"  용접: {(data[2] == 1 ? "진행중" : "대기")}");
            Console.WriteLine($"  전류: {data[3]}A (설정: {data[5]}A)");
            Console.WriteLine($"  전압: {data[4] * 0.1:F1}V (설정: {data[6] * 0.1:F1}V)");

            // PATH 정보
            Console.WriteLine("\n[PATH 정보]");
            Console.WriteLine($"  현재 PATH: {data[8]}");
            Console.WriteLine($"  전체 진행: {data[10]}/{data[11]}");

            // 에러 확인
            if (data[14] == 1)
            {
                Console.WriteLine("\n⚠️  로봇 에러 발생!");
            }
            if (status[3] == 1)
            {
                Console.WriteLine("⚠️  보안 정지 활성화!");
            }
            if (status[4] == 1)
            {
                Console.WriteLine("⚠️  비상 정지 활성화!");
            }
        }

        public void Stop()
        {
            running = false;
        }
    }

    // 사용 예제
    class Program
    {
        static void Main(string[] args)
        {
            WeldingMonitor monitor = new WeldingMonitor("192.168.1.7");
            
            try
            {
                monitor.StartMonitoring(2000);  // 2초마다 업데이트
            }
            catch (Exception ex)
            {
                Console.WriteLine($"연결 실패: {ex.Message}");
            }

            Console.ReadKey();
        }
    }
}

const Modbus = require('jsmodbus');
const net = require('net');

// TCP 소켓 생성
const socket = new net.Socket();
const client = new Modbus.client.TCP(socket, 1);  // unit ID = 1

// 연결
socket.connect({
    host: '192.168.1.7',
    port: 502
});

socket.on('connect', async () => {
    console.log('✓ Modbus TCP 연결 성공\n');

    try {
        // 1. 로봇 → 펜던트 데이터 읽기
        const readResult = await client.readInputRegisters(128, 33);
        const registers = readResult.response.body.values;

        console.log('=== 로봇 상태 ===');
        console.log(`하트비트: ${registers[0]}`);
        console.log(`프로그램 동작: ${registers[1] ? 'ON' : 'OFF'}`);
        console.log(`용접 상태: ${registers[2] ? '용접중' : '대기'}`);
        console.log(`전류: ${registers[3]}A`);
        console.log(`전압: ${registers[4] * 0.1}V`);

        // 2. 작업 모드 설정 (자동 모드)
        await client.writeSingleRegister(162, 2);
        console.log('\n✓ 작업 모드: 자동 모드로 설정');

        // 3. 용접 조건 설정
        await client.writeMultipleRegisters(231, [255, 180]);  // 전압 25.5V, 전류 180A
        console.log('✓ 용접 조건 설정 완료');

    } catch (err) {
        console.error('✗ 에러:', err);
    } finally {
        socket.end();
    }
});

socket.on('error', (err) => {
    console.error('연결 에러:', err);
});

6. 데이터 로깅 예제

import csv
from datetime import datetime
from pymodbus.client import ModbusTcpClient
import time

class WeldingLogger:
    """용접 데이터 로깅 클래스"""
    
    def __init__(self, robot_ip, log_file='welding_log.csv'):
        self.robot_ip = robot_ip
        self.log_file = log_file
        self.client = None
        
        # CSV 헤더 작성
        with open(self.log_file, 'w', newline='', encoding='utf-8') as f:
            writer = csv.writer(f)
            writer.writerow([
                'Timestamp', 'Heartbeat', 'Program_Running', 'Welding_Active',
                'Current_FB', 'Voltage_FB', 'Current_Set', 'Voltage_Set',
                'Current_Path', 'Total_Path', 'Robot_Error', 'Emergency_Stop'
            ])
    
    def connect(self):
        self.client = ModbusTcpClient(self.robot_ip, port=502)
        return self.client.connect()
    
    def log_data(self):
        """데이터 읽기 및 로깅"""
        # 로봇 → 펜던트 데이터
        result = self.client.read_input_registers(address=128, count=33, unit=1)
        if result.isError():
            return False
        
        data = result.registers
        
        # 안전 상태
        result = self.client.read_input_registers(address=262, count=1, unit=1)
        emergency = result.registers[0] if not result.isError() else 0
        
        # CSV에 기록
        with open(self.log_file, 'a', newline='', encoding='utf-8') as f:
            writer = csv.writer(f)
            writer.writerow([
                datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
                data[0],   # Heartbeat
                data[1],   # Program Running
                data[2],   # Welding Active
                data[3],   # Current Feedback
                data[4] * 0.1,  # Voltage Feedback
                data[5],   # Current Setpoint
                data[6] * 0.1,  # Voltage Setpoint
                data[8],   # Current Path
                data[11],  # Total Path
                data[14],  # Robot Error
                emergency  # Emergency Stop
            ])
        
        return True
    
    def start_logging(self, interval=1.0):
        """로깅 시작"""
        print(f"데이터 로깅 시작: {self.log_file}")
        print("Ctrl+C로 종료\n")
        
        try:
            while True:
                if self.log_data():
                    print(f"[{datetime.now().strftime('%H:%M:%S')}] 데이터 기록 완료")
                else:
                    print(f"[{datetime.now().strftime('%H:%M:%S')}] 데이터 읽기 실패")
                
                time.sleep(interval)
        
        except KeyboardInterrupt:
            print("\n\n로깅 종료")
    
    def disconnect(self):
        if self.client:
            self.client.close()

# 사용 예제
if __name__ == "__main__":
    logger = WeldingLogger('192.168.1.7', 'welding_log.csv')
    
    if logger.connect():
        print("✓ 연결 성공")
        logger.start_logging(interval=0.5)  # 0.5초마다 로깅
        logger.disconnect()
    else:
        print("✗ 연결 실패")
